3. 数据清洗

df = df.drop_duplicates() df["age"] = df["age"].fillna(df["age"].median()) df["apply_date"] = pd.to_datetime(df["apply_date"], errors="coerce") df = df[(df["age"] >= 18) & (df["age"] <= 80)]

4. 数据分布与类别不平衡

print(df["label"].value_counts()) print(df["label"].value_counts(normalize=True)) print(df.groupby("label")["score"].agg(["count","mean","median"]))

5. 手工计算混淆矩阵指标

TP, FP, TN, FN = 70, 10, 110, 30 precision = TP / (TP + FP) recall = TP / (TP + FN) accuracy = (TP + TN) / (TP + FP + TN + FN) f1 = 2 * precision * recall / (precision + recall) print(accuracy, precision, recall, f1)
